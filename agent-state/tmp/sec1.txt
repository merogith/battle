            q.textContent = line;
        }

        function cityIndexFromEventIndex(ei) {
            for (let i = ei; i >= 0; i--) {
                const row = STORY_EVENTS_RAW[i];
                if (row && row[1] === 'City') return getCityIndex(row);
            }
            return 0;
        }

        function lastStoryCityEventIndexAtOrBefore(ei) {
            const cap = Math.min(Math.max(0, ei | 0), STORY_EVENTS_RAW.length - 1);
            for (let i = cap; i >= 0; i--) {
                const row = STORY_EVENTS_RAW[i];
                if (row && row[1] === 'City') return i;
            }
            return 0;
        }

        function getStoryDisplayTownNameForCityRow(cityEv) {
            if (!cityEv) return 'Town';
            const rawName = cityEv[2] ? String(cityEv[2]) : '';
            const cityIdx = rawName.startsWith('City') ? parseInt(rawName.slice(4), 10) : -1;
            if (cityIdx === 0) return 'Pallet Town';
            if (cityIdx >= 1 && cityIdx <= 8) {
                const leaderEvIdx = GYM_CITY_LEADER_EVENT[cityIdx];
                const leaderName = sm.trainerAssignments && sm.trainerAssignments[leaderEvIdx];
                return (leaderName && GYM_LEADER_CITY_NAMES[leaderName]) || ('City ' + cityIdx);
            }
            if (cityIdx === 9) return 'Pokémon League';
            return rawName || 'Town';
        }

        function _storyDifficulty() {
            return (sm && sm.storyDifficulty) || 'normal';
        }
        function _storyDifficultyIsCasual(diff) {
            const d = diff || _storyDifficulty();
            return d === 'veryeasy' || d === 'easy';
        }
        // Retreat-to-city after a loss is free on Normal and below. Only
        // Hard / Very Hard charge the half-gold fee.
        function _storyDifficultyHasFreeRetreat(diff) {
            const d = diff || _storyDifficulty();
            return d === 'veryeasy' || d === 'easy' || d === 'normal';
        }
        function _storyGoldFeeAppliesForRetreat() {
            return !_storyDifficultyHasFreeRetreat();
        }

        function _storyCalcRetreatGoldFee() {
            if (!_storyGoldFeeAppliesForRetreat()) return 0;
            const g = Math.max(0, sm.gold | 0);
            if (g <= 0) return 0;
            const half = g / 2;
            const step100 = Math.ceil(half / 100) * 100;
            const raw = Math.max(100, step100);
            return Math.min(raw, g);
        }

        function _storyFullHealPartySlots() {
            (sm.team || []).forEach(slot => {
                if (!slot) return;
                slot.currentHp = null;
                slot.status = null;
                slot.statusTurns = 0;
                slot.pp = null;
                slot.consumedItem = false;
                // PokeCenter / iconic-fight resets also clear fatigue. (Tiredness lives on the
                // build, not the slot, so wipe it there too — applies to every party member
                // regardless of whether they were in the last fight.)
                if (slot.build && typeof slot.build === 'object') slot.build.tired = 0;
            });
        }

        // List of event names that auto-heal and reset fatigue before+after the fight.
        // Wild routes and randomized trainer encounters (Basic / Elite Trainer events) leave
        // fatigue on the survivors. Gym leaders, league, mystery, boss arc, rival, and Pit
        // fights are all "iconic" — full reset on entry, full reset on exit.
        function _storyEventIsIconicFight(eventName) {
            if (!eventName) return false;
            const s = String(eventName);
            return /^Gym Leader [1-8]$/.test(s)
                || /^Gym Trainer/i.test(s)
                || /^E[1-4]$/.test(s)
                || s === 'Champion'
                || s === 'Rival'
                || /Mystery/i.test(s)
                || /^Boss/i.test(s)
                || s === 'Pit'
                || s === 'Crucible'
                || s === 'Frontier';
        }

        // Apply (or clear) fatigue on every active team member. Bench mons are *also*
        // touched on iconic-clear so a player can't game it by rotating to the bench.
        function _storyApplyTiredness(eventName) {
            if (!sm.team || !sm.team.length) return;
            if (_storyEventIsIconicFight(eventName)) {
                // Iconic fights wipe stacks (already wiped on entry — this is the post-clear).
                sm.team.forEach(slot => { if (slot && slot.build) slot.build.tired = 0; });
                return;
            }
            // Non-iconic fight: +1 stack to every party member that holds a build.
            sm.team.forEach(slot => {
                if (!slot || !slot.build) return;
                slot.build.tired = Math.min(3, ((slot.build.tired | 0) + 1));
            });
            // Defer the first-time intro to after the victory overlay dismisses by setting
            // a one-shot queue flag — `afterBattleReturn()` picks it up and fires the modal
            // once the city screen is back up.
            try {
                if (!sm.flags) sm.flags = {};
                if (!sm.flags.seenTirednessIntro) sm.flags._tirednessIntroQueued = true;
            } catch (e) {}
        }

        function _storyClearTirednessForIconic() {
            if (!sm.team) return;
            sm.team.forEach(slot => { if (slot && slot.build) slot.build.tired = 0; });
        }
